182void modbusStartup(){
183 // Initialize RS485 control pins
184 pinMode(MAX485_RE_NEG, OUTPUT);
185 pinMode(MAX485_DE, OUTPUT);
186 digitalWrite(MAX485_RE_NEG, LOW);
187 digitalWrite(MAX485_DE, LOW);
188 Serial.println("ESP32 Modbus RTU Communication Initializing...");
189
190 // Configure UART2 for Modbus communication
191 Serial2.begin(BAUD_RATE, SERIAL_8N1, RX_PIN, TX_PIN);
192
193 // Initialize Modbus master with slave ID and UART2
194 modbus.begin(MODBUS_SLAVE_ID, Serial2);
195
196 // Set pre- and post-transmission callbacks for RS485 control
197 modbus.preTransmission(preTransmission);
198 modbus.postTransmission(postTransmission);
199
200 Serial.println("Modbus RTU Communication Initialized Successfully");
201}
